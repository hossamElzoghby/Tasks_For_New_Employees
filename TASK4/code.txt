package routines;

import com.temenos.t24.api.complex.eb.templatehook.TransactionContext;
import com.temenos.t24.api.hook.system.RecordLifecycle;

public class IDRTN extends RecordLifecycle {
    @Override
    public String checkId(String currentRecordId, TransactionContext transactionContext) {
        if(!currentRecordId.matches("\\d{3}-\\d{2}-\\d{2}[A-Za-z]{2}")){
            throw new Error(currentRecordId+"INVALID FORMAT");
        }
        int yy= Integer.parseInt(currentRecordId.substring(4,6));
        int bb=Integer.parseInt(currentRecordId.substring(7,9));
        if(yy>bb){
            throw new Error(currentRecordId+"INVALID FORMAT");
        }
        return currentRecordId+"SYSTEM";
    }
}

package routines;

import com.temenos.t24.api.hook.system.RecordLifecycle;
import com.temenos.t24.api.records.stmbsctrain.StMbscTrainRecord;
import com.temenos.api.TStructure;
import com.temenos.t24.api.complex.eb.templatehook.TransactionContext;




public class CHECKREC extends RecordLifecycle{

    @Override
    public void defaultFieldValues(String application, String currentRecordId, TStructure currentRecord,
            TStructure unauthorisedRecord, TStructure liveRecord,TransactionContext transactionContext) {
        
        StMbscTrainRecord rec = new StMbscTrainRecord (currentRecord);
        
        String tenor=currentRecordId.substring(0,3);
        String frequency=currentRecordId.substring(4,6);
        String bb=currentRecordId.substring(7,9);
        String letters=currentRecordId.substring(9,11);
        
        rec.setTenor(tenor);
        rec.setFrequency(frequency);
        rec.setMixedType(tenor+frequency+bb+letters);
        
        if(letters.equals("MX")){
            rec.setMixedType("YES");
    }

    }
}

public class VALIDREC extends RecordLifecycle{
    @Override
    public void defaultFieldValues(String application, String currentRecordId, TStructure currentRecord,
            TStructure unauthorisedRecord, TStructure liveRecord,TransactionContext transactionContext) {
        
        StMbscTrainRecord rec = new StMbscTrainRecord (currentRecord);
        
       if(rec.getFloating().getValue().equals("YES")){
           if(rec.getBasicKey().getValue().isEmpty()){
               throw new Error("BASIC.KEY, SHOULD BE MANDATORY");
           }
           if(rec.getIntSpread().getValue().isEmpty()){
               throw new Error("INT.SPREAD, SHOULD BE MANDATORY");
           }
    }
       if(rec.getAccumlated().getValue().equals("YES")){
           if(rec.getRate().getValue().isEmpty()){
               throw new Error("RATE, SHOULD BE MANDATORY");
           }
           if(rec.getIntRate().getValue().isEmpty()){
               throw new Error("INT.RATE, SHOULD BE MANDATORY");
           }
       }
    }
}
